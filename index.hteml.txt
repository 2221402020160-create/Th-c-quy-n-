        // Web Audio API for sound effects
        const audioCtx = new (window.AudioContext || window.webkitAudioContext)();

        function playSound(type) {
            if (audioCtx.state === 'suspended') {
                audioCtx.resume();
            }
            const osc = audioCtx.createOscillator();
            const gain = audioCtx.createGain();
            osc.connect(gain);
            gain.connect(audioCtx.destination);

            const now = audioCtx.currentTime;

            if (type === 'tick') {
                osc.type = 'triangle';
                osc.frequency.setValueAtTime(400, now);
                osc.frequency.exponentialRampToValueAtTime(800, now + 0.05);
                gain.gain.setValueAtTime(0.15, now);
                gain.gain.linearRampToValueAtTime(0.01, now + 0.05);
                osc.start(now);
                osc.stop(now + 0.05);
            } else if (type === 'win') {
                osc.type = 'sine';
                osc.frequency.setValueAtTime(523.25, now); // C5
                osc.frequency.setValueAtTime(659.25, now + 0.1); // E5
                osc.frequency.setValueAtTime(783.99, now + 0.2); // G5
                osc.frequency.setValueAtTime(1046.50, now + 0.3); // C6
                gain.gain.setValueAtTime(0.3, now);
                gain.gain.linearRampToValueAtTime(0.01, now + 0.6);
                osc.start(now);
                osc.stop(now + 0.6);
            } else if (type === 'click') {
                osc.type = 'sine';
                osc.frequency.setValueAtTime(600, now);
                gain.gain.setValueAtTime(0.1, now);
                gain.gain.linearRampToValueAtTime(0.01, now + 0.03);
                osc.start(now);
                osc.stop(now + 0.03);
            }
        }

        // Initialize Background Stars
        function initStars() {
            const container = document.getElementById('starsContainer');
            container.innerHTML = '';
            for (let i = 0; i < 40; i++) {
                const star = document.createElement('div');
                star.className = 'bg-star';
                const size = Math.random() * 4 + 2;
                star.style.width = `${size}px`;
                star.style.height = `${size}px`;
                star.style.top = `${Math.random() * 100}%`;
                star.style.left = `${Math.random() * 100}%`;
                star.style.setProperty('--duration', `${Math.random() * 2 + 1}s`);
                container.appendChild(star);
            }
        }

        // Load data from LocalStorage or Default
        function loadCardsData() {
            const saved = localStorage.getItem('school_3d_cards');
            if (saved) {
                try {
                    cardsData = JSON.parse(saved);

                    cardsData = [...DEFAULT_CARDS];
                }

        }

        // Build 3D Carousel Elements
        function renderCarousel() {
            const stage = document.getElementById('carouselStage');
            stage.innerHTML = '';
            cardElements = [];

            cardsData.forEach((item, index) => {

        // Update positions along an Ellipse in 3D Space
        function updateCarouselPositions() {
            const count = cardElements.length;
            const isMobile = window.innerWidth < 768;
            
            // Ellipse radii
            const rx = isMobile ? 140 : 280; 
            const ry = isMobile ? 50 : 80;

            cardElements.forEach((card, i) => {
                const angle = currentAngle + (i * (2 * Math.PI / count));
                
                const x = Math.sin(angle) * rx;
                const y = Math.cos(angle) * ry;
                
                // Depth scale and z-index calculation
                const scale = (y + ry + 100) / (2 * ry + 100) * 0.45 + 0.65;
                const zIndex = Math.round((y + ry) * 10);
                const opacity = (y + ry) / (2 * ry) * 0.5 + 0.5;

                card.style.transform = `translate3d(${x}px, ${y}px, 0px) scale(${scale})`;
                card.style.zIndex = zIndex;
                card.style.opacity = opacity;
            });
        }

        // Idle Slow Elliptic Orbit Animation
        function idleAnimate() {
            if (!isSpinning) {
                currentAngle += 0.003;
                updateCarouselPositions();
            }
            animationFrameId = requestAnimationFrame(idleAnimate);
        }

        // Spin & Shuffle Trigger
        function startSpinning() {
            if (isSpinning) return;
            isSpinning = true;
            playSound('click');

            const btn = document.getElementById('btnShuffle');
            btn.disabled = true;
            btn.classList.add('opacity-50', 'cursor-not-allowed');

            let speed = 0.25;
            let duration = 3000; // 3 seconds spin
            let startTime = performance.now();
            let lastTickAngle = currentAngle;

            function spinStep(now) {
                const elapsed = now - startTime;

                if (elapsed < duration) {
                    currentAngle += speed;
                    
                    // Sound tick during rotation
                    if (Math.abs(currentAngle - lastTickAngle) > 0.4) {
                        playSound('tick');
                        lastTickAngle = currentAngle;
                    }

                    updateCarouselPositions();
                    requestAnimationFrame(spinStep);
                } else {
                    // Pick Random Card
                    const randomIndex = Math.floor(Math.random() * cardsData.length);
                    selectSpecificCard(randomIndex);

                    isSpinning = false;
                    btn.disabled = false;
                    btn.classList.remove('opacity-50', 'cursor-not-allowed');
                }
            }

            requestAnimationFrame(spinStep);
        }

        // Focus & Show Winner Card
        function selectSpecificCard(index) {
            playSound('win');
            const item = cardsData[index];

            // Populate Modal Data
            document.getElementById('modalCardBadge').innerText = `Thẻ số ${index + 1}`;
            document.getElementById('modalCardIcon').innerText = item.icon || '🎁';
            document.getElementById('modalCardTitle').innerText = item.title;
            document.getElementById('modalCardReward').innerText = `🎁 ${item.reward}`;

            // Trigger Fireworks / Confetti
            confetti({
                particleCount: 120,
                spread: 80,
                origin: { y: 0.6 }
            });

            // Show Modal with smooth animation
            const modal = document.getElementById('winnerModal');
            const box = document.getElementById('winnerBox');
            modal.classList.remove('hidden');
            setTimeout(() => {
                box.classList.remove('scale-95', 'opacity-0');
                box.classList.add('scale-100', 'opacity-100');
            }, 10);
        }

        function closeWinnerModal() {

            box.classList.remove('scale-100', 'opacity-100');
            box.classList.add('scale-95', 'opacity-0');
            setTimeout(() => {
                document.getElementById('winnerModal').classList.add('hidden');
            }, 200);
        }

                        <span class="text-xs font-extrabold text-yellow-300 w-12">#${idx + 1}</span>
                        <input type="text" value="${item.icon || '⭐'}" class="w-12 text-center bg-purple-950 border border-pink-400 rounded-lg p-1.5 text-xl" text-center onchange="updateItemIcon(${idx}, this.value)">
                        <input type="text" value="${item.title}" placeholder="Tên thẻ" class="flex-1 md:w-40 bg-purple-950 border border-pink-400 rounded-lg px-3 py-1.5 text-sm font-bold text-white" onchange="updateItemTitle(${idx}, this.value)">
                    </div>
                    <div class="flex-1 w-full flex gap-2 items-center">
                        <input type="text" value="${item.reward}" placeholder="Phần thưởng" class="flex-1 bg-purple-950 border border-pink-400 rounded-lg px-3 py-1.5 text-sm text-yellow-200" onchange="updateItemReward(${idx}, this.value)">
                        <button onclick="removeCardItem(${idx})" class="bg-red-500/80 hover:bg-red-600 text-white p-2 rounded-lg text-xs">
                            <i class="fa-solid fa-trash"></i>
                        </button>
                    </div>
                `;
                container.appendChild(row);
            });
        }

        function updateItemIcon(index, val) { cardsData[index].icon = val; }
        function updateItemTitle(index, val) { cardsData[index].title = val; }
        function updateItemReward(index, val) { cardsData[index].reward = val; }

        function addNewCardItem() {
            cardsData.push({
                id: Date.now(),
                title: `Thẻ Mới ${cardsData.length + 1}`,
                reward: "Một vỗ tay khen ngợi!",
                icon: "🌟"
            });
            renderEditForm();
        }

        function removeCardItem(index) {
            if (cardsData.length <= 3) {
                alert("Cần giữ ít nhất 3 thẻ để vòng xoay đẹp mắt!");
                return;
            }
            cardsData.splice(index, 1);
            renderEditForm();
        }

        function saveRewards() {
            localStorage.setItem('school_3d_cards', JSON.stringify(cardsData));
            renderCarousel();
            closeEditModal();
        }

        function resetToDefault() {
            if (confirm("Khôi phục về danh sách thẻ bài mặc định?")) {
                cardsData = JSON.parse(JSON.stringify(DEFAULT_CARDS));
                localStorage.removeItem('school_3d_cards');
                renderEditForm();
                renderCarousel();
            }
        }

        // Window resize listener
        window.addEventListener('resize', updateCarouselPositions);

        // On Load Init
        window.onload = function() {
            initStars();
            loadCardsData();
            renderCarousel();
            idleAnimate();
        };
    </script>
